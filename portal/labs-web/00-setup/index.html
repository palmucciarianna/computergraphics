Ciclo di Rendering GLFW (Game Loop)
while (!glfwWindowShouldClose(window)) {
  // 1. Input
  processInput(window);

  // 2. Render
  glClearColor(r, g, b, 1.0f);
  glClear(GL_COLOR_BUFFER_BIT);

  // 3. Swap Buffers & Poll Events
  glfwSwapBuffers(window);
  glfwPollEvents();
}
